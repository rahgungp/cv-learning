Praktikum 02: Image Processing dengan OpenCV

Color space conversion, histogram equalization, dan augmentasi data menggunakan OpenCV dan torchvision.

▶ Buka di Google Colab

Step 1: Konversi Color Space

Python
import cv2
import numpy as np
import matplotlib.pyplot as plt

img_bgr = cv2.imread("sample.jpg")

# Konversi ke berbagai color space
img_rgb  = cv2.cvtColor(img_bgr, cv2.COLOR_BGR2RGB)
img_hsv  = cv2.cvtColor(img_bgr, cv2.COLOR_BGR2HSV)
img_gray = cv2.cvtColor(img_bgr, cv2.COLOR_BGR2GRAY)
img_lab  = cv2.cvtColor(img_bgr, cv2.COLOR_BGR2LAB)

# Tampilkan semua
fig, axes = plt.subplots(2, 2, figsize=(10, 8))
axes[0,0].imshow(img_rgb);                       axes[0,0].set_title("RGB")
axes[0,1].imshow(img_hsv[:,:,0], cmap="hsv");    axes[0,1].set_title("HSV - Hue channel")
axes[1,0].imshow(img_gray, cmap="gray");          axes[1,0].set_title("Grayscale")
axes[1,1].imshow(img_lab[:,:,0], cmap="gray");    axes[1,1].set_title("LAB - L channel")
for ax in axes.flat: ax.axis("off")
plt.tight_layout(); plt.show()

Step 2: Histogram & Equalization

Python
import cv2
import matplotlib.pyplot as plt
import numpy as np

img_bgr  = cv2.imread("sample.jpg")
img_gray = cv2.cvtColor(img_bgr, cv2.COLOR_BGR2GRAY)

# Histogram sebelum equalization
hist_before = cv2.calcHist([img_gray], [0], None, [256], [0, 256])

# Histogram Equalization
img_eq = cv2.equalizeHist(img_gray)
hist_after = cv2.calcHist([img_eq], [0], None, [256], [0, 256])

# CLAHE — lebih halus dari equalizeHist
clahe = cv2.createCLAHE(clipLimit=2.0, tileGridSize=(8, 8))
img_clahe = clahe.apply(img_gray)

fig, axes = plt.subplots(2, 3, figsize=(15, 8))
for img, title, ax in zip(
    [img_gray, img_eq, img_clahe,
     hist_before, hist_after, None],
    ["Original Gray", "Equalized", "CLAHE", "Hist Before", "Hist After", ""],
    axes.flat
):
    if img is not None:
        if len(img.shape) == 1 or (len(img.shape)==2 and img.shape[1]<10):
            ax.plot(img, color="navy"); ax.set_title(title)
        else:
            ax.imshow(img, cmap="gray"); ax.set_title(title); ax.axis("off")
plt.tight_layout(); plt.show()

Step 3: Data Augmentation

Python
from torchvision import transforms
from PIL import Image
import matplotlib.pyplot as plt
import numpy as np

img = Image.open("sample.jpg")

# Definisikan augmentasi
augmentations = [
    ("Original",            transforms.Compose([transforms.Resize((224,224))])),
    ("Horizontal Flip",     transforms.Compose([transforms.Resize((224,224)), transforms.RandomHorizontalFlip(p=1.0)])),
    ("Rotation 30°",        transforms.Compose([transforms.Resize((224,224)), transforms.RandomRotation(30)])),
    ("Color Jitter",        transforms.Compose([transforms.Resize((224,224)), transforms.ColorJitter(0.5, 0.5, 0.5, 0.1)])),
    ("Random Crop",         transforms.Compose([transforms.RandomResizedCrop(224, scale=(0.6,1.0))])),
    ("Grayscale",           transforms.Compose([transforms.Resize((224,224)), transforms.Grayscale(num_output_channels=3)])),
]

fig, axes = plt.subplots(2, 3, figsize=(12, 8))
for ax, (title, transform) in zip(axes.flat, augmentations):
    out = transform(img)
    ax.imshow(np.array(out))
    ax.set_title(title)
    ax.axis("off")
plt.tight_layout(); plt.show()
print("✅ Augmentasi berhasil!")
