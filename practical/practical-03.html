Praktikum 03: Filtering & Edge Detection

Terapkan berbagai filter pada gambar nyata: Gaussian blur, Canny edge detection, dan Sobel operator.

▶ Buka di Google Colab

Step 1: Filter Dasar

Python
import cv2
import numpy as np
import matplotlib.pyplot as plt

img = cv2.imread("sample.jpg", cv2.IMREAD_GRAYSCALE)

# Berbagai filter blur
gaussian = cv2.GaussianBlur(img, (11, 11), 0)
median   = cv2.medianBlur(img, 11)
bilateral = cv2.bilateralFilter(img, 9, 75, 75)

# Sharpen
kernel_sharp = np.array([[0, -1, 0], [-1, 5, -1], [0, -1, 0]], np.float32)
sharpened = cv2.filter2D(img, -1, kernel_sharp)

fig, axes = plt.subplots(2, 3, figsize=(14, 8))
for ax, (image, title) in zip(axes.flat, [
    (img, "Original"), (gaussian, "Gaussian Blur"),
    (median, "Median Blur"), (bilateral, "Bilateral Filter"),
    (sharpened, "Sharpened"), (np.clip(img.astype(int)-gaussian.astype(int)+128, 0, 255).astype(np.uint8), "Unsharp Mask")
]):
    ax.imshow(image, cmap="gray"); ax.set_title(title); ax.axis("off")
plt.tight_layout(); plt.show()

Step 2: Edge Detection

Python
import cv2
import numpy as np
import matplotlib.pyplot as plt

img = cv2.imread("sample.jpg", cv2.IMREAD_GRAYSCALE)
blurred = cv2.GaussianBlur(img, (5, 5), 0)

# Canny edge detection
edges_low  = cv2.Canny(blurred, 30, 100)
edges_mid  = cv2.Canny(blurred, 80, 160)
edges_high = cv2.Canny(blurred, 150, 250)

# Sobel
sobel_x = cv2.Sobel(img, cv2.CV_64F, 1, 0, ksize=3)
sobel_y = cv2.Sobel(img, cv2.CV_64F, 0, 1, ksize=3)
sobel_mag = np.sqrt(sobel_x**2 + sobel_y**2)
sobel_mag = np.uint8(255 * sobel_mag / sobel_mag.max())

# Laplacian
laplacian = cv2.Laplacian(blurred, cv2.CV_64F)
laplacian = np.uint8(np.abs(laplacian))

fig, axes = plt.subplots(2, 3, figsize=(14, 8))
for ax, (image, title) in zip(axes.flat, [
    (img, "Original"), (edges_low, "Canny (Low)"),
    (edges_mid, "Canny (Mid)"), (sobel_x, "Sobel X"),
    (sobel_mag, "Sobel Magnitude"), (laplacian, "Laplacian")
]):
    ax.imshow(image, cmap="gray"); ax.set_title(title); ax.axis("off")
plt.tight_layout(); plt.show()
