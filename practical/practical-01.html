Praktikum 01: Eksplorasi Gambar dengan Python

Hands-on: baca gambar, lihat representasi array, pisahkan channel RGB, dan buat visualisasi dasar dengan OpenCV + PIL + Matplotlib.

 Buka di Google Colab
📋 Tujuan Praktikum
  • Membaca dan menampilkan gambar menggunakan PIL dan OpenCV
  • Memahami representasi array NumPy untuk gambar
  • Memisahkan dan menampilkan channel RGB secara terpisah
  • Melakukan konversi ke grayscale dan melihat nilai piksel
  • Membuat subplot multi-panel dengan Matplotlib

Step 1: Baca Gambar dengan PIL

Python
from PIL import Image
import numpy as np
import matplotlib.pyplot as plt

# Download gambar sample (di Colab)
# !wget -O sample.jpg https://upload.wikimedia.org/wikipedia/commons/thumb/3/3a/Cat03.jpg/300px-Cat03.jpg

# Buka gambar dengan PIL
img_pil = Image.open("sample.jpg")

print(f"Format : {img_pil.format}")      # JPEG
print(f"Mode   : {img_pil.mode}")        # RGB
print(f"Size   : {img_pil.size}")        # (width, height)

# Konversi ke NumPy array
img_arr = np.array(img_pil)
print(f"\nShape  : {img_arr.shape}")     # (H, W, 3)
print(f"Dtype  : {img_arr.dtype}")       # uint8
print(f"Min    : {img_arr.min()}")       # 0
print(f"Max    : {img_arr.max()}")       # 255

# Tampilkan gambar
plt.figure(figsize=(6, 5))
plt.imshow(img_arr)
plt.title(f"Gambar: {img_arr.shape}")
plt.axis('off')
plt.show()

Step 2: Eksplorasi Channel RGB

Python
import numpy as np
import matplotlib.pyplot as plt
from PIL import Image

img = np.array(Image.open("sample.jpg"))

# Pisahkan channel
R = img[:, :, 0]   # Channel merah
G = img[:, :, 1]   # Channel hijau
B = img[:, :, 2]   # Channel biru

# Buat gambar per channel (dengan nol di channel lain)
img_R = np.zeros_like(img); img_R[:, :, 0] = R
img_G = np.zeros_like(img); img_G[:, :, 1] = G
img_B = np.zeros_like(img); img_B[:, :, 2] = B

# Tampilkan 4 panel
fig, axes = plt.subplots(1, 4, figsize=(16, 4))
titles  = ["Original (RGB)", "Red Channel", "Green Channel", "Blue Channel"]
images  = [img, img_R, img_G, img_B]
for ax, image, title in zip(axes, images, titles):
    ax.imshow(image)
    ax.set_title(title)
    ax.axis('off')
plt.tight_layout()
plt.show()

# Statistik per channel
for name, ch in zip(['R', 'G', 'B'], [R, G, B]):
    print(f"Channel {name}: min={ch.min()}, max={ch.max()}, mean={ch.mean():.1f}")

Step 3: Konversi Grayscale dan Akses Piksel

Python
import cv2
import numpy as np
import matplotlib.pyplot as plt

# OpenCV membaca dalam BGR!
img_bgr = cv2.imread("sample.jpg")
img_rgb = cv2.cvtColor(img_bgr, cv2.COLOR_BGR2RGB)

# Konversi ke grayscale
img_gray = cv2.cvtColor(img_bgr, cv2.COLOR_BGR2GRAY)

print(f"Grayscale shape: {img_gray.shape}")    # (H, W) — 2D!
print(f"Grayscale dtype: {img_gray.dtype}")    # uint8

# Formula grayscale (luminance weighting):
# Y = 0.299*R + 0.587*G + 0.114*B

# Akses piksel tunggal
piksel_rgb  = img_rgb[100, 150]        # array([R, G, B])
piksel_gray = img_gray[100, 150]       # nilai integer tunggal
print(f"\nPiksel [100,150] RGB  : {piksel_rgb}")
print(f"Piksel [100,150] Gray : {piksel_gray}")

# Akses region (bounding box)
roi = img_rgb[50:150, 100:200]   # Baris 50-150, Kolom 100-200
print(f"\nROI shape: {roi.shape}")

# Tampilkan
fig, axes = plt.subplots(1, 3, figsize=(14, 4))
axes[0].imshow(img_rgb);    axes[0].set_title("RGB");       axes[0].axis('off')
axes[1].imshow(img_gray, cmap='gray'); axes[1].set_title("Grayscale"); axes[1].axis('off')
axes[2].imshow(roi);        axes[2].set_title("ROI");       axes[2].axis('off')
plt.tight_layout()
plt.show()

Step 4: Simpan Gambar

Python
import cv2
from PIL import Image
import numpy as np

img_bgr = cv2.imread("sample.jpg")

# Simpan dengan OpenCV
cv2.imwrite("output_gray.jpg", cv2.cvtColor(img_bgr, cv2.COLOR_BGR2GRAY))
cv2.imwrite("output_small.jpg", cv2.resize(img_bgr, (128, 128)))

# Simpan dengan PIL
img_pil = Image.open("sample.jpg")
img_pil.save("output_pil.png")    # PNG (lossless)

# Simpan dengan kualitas tertentu
img_pil.save("output_q80.jpg", "JPEG", quality=80)

print("✅ Semua gambar tersimpan!")
✅ Tantangan
  1. Hitung nilai piksel rata-rata untuk keseluruhan gambar
  2. Cari piksel paling terang (nilai tertinggi) dan koordinatnya
  3. Buat gambar "negatif" — balik nilai piksel (255 - nilai_piksel)
  4. Crop region wajah dari gambar (jika ada) dan simpan sebagai file baru
