Praktikum 04: Membangun CNN dengan PyTorch

Implementasi CNN dari nol dengan PyTorch, latih pada dataset CIFAR-10, dan analisis feature maps.

▶ Buka di Google Colab

Step 1: CNN sederhana

Python
import torch
import torch.nn as nn
import torch.optim as optim
from torchvision import datasets, transforms
from torch.utils.data import DataLoader

# Dataset CIFAR-10
transform = transforms.Compose([
    transforms.ToTensor(),
    transforms.Normalize((0.5, 0.5, 0.5), (0.5, 0.5, 0.5))
])
trainset = datasets.CIFAR10(root="./data", train=True,  download=True, transform=transform)
testset  = datasets.CIFAR10(root="./data", train=False, download=True, transform=transform)
trainloader = DataLoader(trainset, batch_size=64, shuffle=True,  num_workers=2)
testloader  = DataLoader(testset,  batch_size=64, shuffle=False, num_workers=2)

classes = ["airplane","automobile","bird","cat","deer","dog","frog","horse","ship","truck"]
print(f"Training: {len(trainset)} | Test: {len(testset)}")

# Arsitektur CNN
class MyCNN(nn.Module):
    def __init__(self):
        super().__init__()
        self.features = nn.Sequential(
            nn.Conv2d(3, 32, 3, padding=1), nn.ReLU(), nn.MaxPool2d(2,2),
            nn.Conv2d(32, 64, 3, padding=1), nn.ReLU(), nn.MaxPool2d(2,2),
        )
        self.classifier = nn.Sequential(
            nn.Flatten(),
            nn.Linear(64*8*8, 256), nn.ReLU(), nn.Dropout(0.5),
            nn.Linear(256, 10)
        )
    def forward(self, x): return self.classifier(self.features(x))

device = "cuda" if torch.cuda.is_available() else "cpu"
model  = MyCNN().to(device)
print(f"Device: {device}")
print(f"Parameters: {sum(p.numel() for p in model.parameters()):,}")

Step 2: Training Loop

Python
# Lanjutan dari cell sebelumnya
import matplotlib.pyplot as plt

criterion = nn.CrossEntropyLoss()
optimizer = optim.Adam(model.parameters(), lr=1e-3)
scheduler = optim.lr_scheduler.StepLR(optimizer, step_size=5, gamma=0.5)

def train_epoch(model, loader, criterion, optimizer):
    model.train(); total_loss = correct = total = 0
    for X, y in loader:
        X, y = X.to(device), y.to(device)
        optimizer.zero_grad()
        out = model(X); loss = criterion(out, y)
        loss.backward(); optimizer.step()
        total_loss += loss.item(); pred = out.argmax(1)
        correct += pred.eq(y).sum().item(); total += y.size(0)
    return total_loss/len(loader), correct/total

def eval_epoch(model, loader, criterion):
    model.eval(); total_loss = correct = total = 0
    with torch.no_grad():
        for X, y in loader:
            X, y = X.to(device), y.to(device)
            out = model(X); loss = criterion(out, y)
            total_loss += loss.item(); pred = out.argmax(1)
            correct += pred.eq(y).sum().item(); total += y.size(0)
    return total_loss/len(loader), correct/total

EPOCHS = 10; train_accs = []; val_accs = []
for epoch in range(1, EPOCHS+1):
    tr_loss, tr_acc = train_epoch(model, trainloader, criterion, optimizer)
    va_loss, va_acc = eval_epoch(model, testloader, criterion)
    scheduler.step()
    train_accs.append(tr_acc); val_accs.append(va_acc)
    print(f"Epoch {epoch:2d} | Train: {tr_loss:.4f} ({tr_acc:.1%}) | Val: {va_loss:.4f} ({va_acc:.1%})")

plt.plot(train_accs, label="Train"); plt.plot(val_accs, label="Val")
plt.xlabel("Epoch"); plt.ylabel("Accuracy"); plt.legend(); plt.title("Training Curve"); plt.show()
